\section{Double Well Potential with MCMC}

Consider the double well potential,
\begin{equation}
  V(x) =  \alpha x^4  - 2x^2 + \frac{1}{\alpha}
\end{equation}
where $x$ is the position of the particle, and we set $m=\hbar=1$ and $\alpha = 0.4$.
See Refs.~\cite{Mittal:2020,Rodgers:2014} for discussions of similar problems.

Use the path integral formulation with imaginary time $\tau$,
\begin{align}
  Z & = \int \mathcal D x(\tau) \exp\left [-\frac{1}{\hbar}\oint^{\tau_b}_{0} L_E(x(\tau)) d\tau \right]]                                                                                                                                                                                        \\
    & =\lim_{\delta\tau\to 0}   \int \cdots \int dx_0\cdots dx_{N-1}  \left (\frac{2\pi \hbar \delta\tau}{m} \right)^{-\frac{N}{2}} \exp\left [-\frac{1}{\hbar} \left(\sum_{i=1}^{N}\frac{m(x_{i} - x_{i-1})^2}{2\delta\tau} + \delta\tau V \left(\frac{x_{i-1}+x_{i}}{2}\right )\right)\right ]
\end{align}
where $L_E(x(\tau)) = \textstyle\frac{m}{2} \left(\frac{dx}{d\tau}\right)^2 + V(x(\tau))$ is the Euclidean Lagrangian, imaginary time is discretized with $N$ increments, $\tau_a = 0$, and $\tau_b = N\delta\tau = \hbar\beta$.
So the probability of a given path $(x_0, x_1, \dots, x_{N-1})$ is
\begin{align}
  p(x_0,\dots, x_{N-1}) & \propto \exp\left [-\frac{1}{\hbar} \left(\sum_{i=1}^{N}\frac{m(x_{i} - x_{i-1})^2}{2\delta\tau} + \delta\tau V \left(\frac{x_{i-1}+x_{i}}{2}\right )\right)\right ]
\end{align}

\begin{problem}
  Evaluate the ground state energy and probability distribution of the particle using Markov chain Monte Carlo with the Metropolis-Hastings algorithm in the large-$\tau_b$ (imaginary time) limit.
  Describe your strategy for determining $\tau_b$, the initial configuration, burn-in steps, hit size, number of sweeps, and thinning (if any).
\end{problem}

\begin{problem}
  Plot the ground state probability distribution and compare it with the expected form.
\end{problem}

\begin{problem}
  Calculate the energy and probability distribution of the particle from the same simulation code for a smaller value of $\tau_b$.
  What is the expected probability distribution in this case?
  What does this correspond to in terms of a statistical mechanics interpretation?
\end{problem}

\section{2D Ising Model with MCMC}

Consider the 2D Ising model in a square lattice $\Lambda$ with $100\times 100$ sites and periodic boundary conditions in the presence of an external magnetic field $B$.
The energy of the system for a given spin configuration $\sigma = \{\sigma_i\}_{i\in \Lambda}$ is
\begin{equation}
  E(\sigma) = -J\sum_{\langle ij\rangle} \sigma_i\sigma_j - B\sum_{i\in \Lambda} \sigma_i,
\end{equation}
where $\langle i j \rangle$ denotes two adjacent sites (with no double counting), $J$ is the spin-spin interaction, and $\sigma_i \in \{ -1, +1\}$ is the spin at site $i$.

The magnetization of the system is
\begin{equation}
  M(\sigma) = \frac{1}{|\Lambda|}\sum_{i\in \Lambda} \sigma_i.
\end{equation}

\begin{problem}
  Use Markov chain Monte Carlo and the Metropolis-Hastings algorithm to simulate the 2D Ising model at different temperatures $T$ and magnetic field strengths $B$.
  Discuss your strategy for determining the initial configuration, burn-in steps, total number of steps, and thinning (if any).
\end{problem}


\begin{problem}
  Plot 1D scans of the magnetization $M$ versus $T$ for fixed $B$ at three different values: $B<0$, $B=0$, and $B>0$.
  Plot 1D scans of the magnetization $M$ versus $B$ for fixed $T$ at three different values: $T<T_C$, $T=T_C$, and $T>T_C$, where $T_C=\frac{2}{\ln(1+\sqrt{2})}\approx 2.269$.
\end{problem}

\begin{problem}
  Putting this all together, draw/describe the phase diagram in $B$ versus $T$ of the 2D Ising model, where the magnetization $M$ is the order parameter.
  Consider discussing first-order phase transitions and critical exponents, hysteresis and metastable states, and/or specific heat and susceptibility.
  See Refs.~\cite{Eastman:2014,Krauss:2018} for relevant discussions.
\end{problem}

\section{Drell--Yan Event Generator with VEGAS}

Consider the Drell--Yan production process at an electron-positron collider, in which an electron and positron collide to produce a virtual photon or a $Z$ boson that then decays into a muon-antimuon pair, $e^+e^-\to Z/\gamma \to \mu^+\mu^-$.

\begin{center}
  \includegraphics[width=0.45\textwidth]{eemumu.pdf}
  \includegraphics[width=0.45\textwidth]{eemumukin.pdf}
\end{center}

As described in lecture and in Ref.~\cite{Papaefstathiou:2014mya}, the differential cross section for center-of-mass energy $E_\text{CM} = \sqrt{\hat s}$ and scattering angle $\theta$ is given by
\begin{equation}
  \label{eq:partonicxs}
  \frac{ d\sigma } { d \Omega }(\hat s, \cos\theta) = \frac{\alpha^2}{4 \hat{s} } \left[ A_0 (\hat s) ( 1  + \cos ^2 \theta ) + A_1(\hat s) \cos \theta \right] \;,
\end{equation}
where $A_0$ and $A_1$ are given by
\begin{align}
  A_0(\hat s) & = Q_e^2 - 2 Q_e V_\mu  V_e ~\chi_1(\hat s) + (A_\mu^2 + V_\mu^2)  (A_e^2 + V_e^2)  ~\chi_2(\hat s)\;, \nonumber \\
  A_1(\hat s) & = -4 Q_e  A_\mu A_e ~\chi_1(\hat s) + 8 A_\mu V_\mu A_e V_e ~ \chi_2(\hat s)\;,
\end{align}
and the $\chi_1$ and $\chi_2$ are given by
\begin{align}
  \chi_1 (\hat{s}) & = \kappa \hat{s} ( \hat{s} - M_Z^2 ) / (  (\hat{s}-M_Z^2)^2 + \Gamma_Z^2 M_Z^2 ) \;, \nonumber \\
  \chi_2 (\hat{s}) & = \kappa^2 \hat{s}^2 / (  (\hat{s}-M_Z^2)^2 + \Gamma_Z^2 M_Z^2 ) \;, \nonumber                 \\
  \kappa           & = \sqrt{2} G_F M_Z^2 / (4 \pi \alpha) \;.
\end{align}

Useful constants are given in the tables below.

\begin{center}
  \begin{tabular}{llll}
    Fermions                       & $Q_f$          & $V_f$                                            & $A_f$          \\ \hline
    $u$, $c$, $t$                  & $+\frac{2}{3}$ & $(+\frac{1}{2} - \frac{4}{3} \sin ^2 \theta_W )$ & $+\frac{1}{2}$ \\
    $d$, $s$, $b$                  & $-\frac{1}{3}$ & $(-\frac{1}{2} - \frac{2}{3} \sin ^2 \theta_W )$ & $-\frac{1}{2}$ \\
    $\nu_e$, $\nu_\mu$, $\nu_\tau$ & $0$            & $\frac{1}{2}$                                    & $+\frac{1}{2}$ \\
    $e$, $\mu$, $\tau$             & $-1$           & $(-\frac{1}{2} + 2 \sin ^2 \theta_W )$           & $-\frac{1}{2}$ \\
  \end{tabular}
\end{center}

\begin{center}
  \begin{tabular}{lll}
    Variable             & Symbol                        & Value                                  \\ \hline
    conversion factor    & GeV$^{-2}\leftrightarrow$\,pb & $3.894\times 10^8$\,pb = 1\,GeV$^{-2}$ \\
    $Z$ boson mass       & $M_Z$                         & 91.188\,GeV                            \\
    $Z$ boson width      & $\Gamma_Z$                    & 2.4414\,GeV                            \\
    QED running coupling & $\alpha$                      & $\frac{1}{132.507}$                    \\

  Note, usually $E_\text{CM}$ is fixed in an electron-positron collider, but we will consider a range of energies, which is similar to the situation at a hadron collider where the partonic center-of-mass energy is not known exactly.

\section{Polymer Collapse with MCMC: A Toy Model of Protein Folding}

A protein is a chain of amino acids that, in water, collapses from an extended coil into a compact structure.
Coarse-grained models replace each amino acid with a single bead, and even a \emph{homopolymer}, in which all beads are identical, captures the first step of folding: as the temperature is lowered, the chain undergoes a transition from a swollen coil to a compact globule~\cite{deGennes:1979}.
(Folding into one specific structure requires different interactions between different beads; see the extensions below.)
This project, introduced in Lecture 21, is adapted from a homework assignment on MCMC for protein folding in CS~2840 (Spring 2025).

Consider a chain of $N$ beads at positions $\vec r_1, \dots, \vec r_N$ in three dimensions.
In reduced units ($\epsilon = \sigma = k_B = 1$), the potential energy is
\begin{equation}
  U(\vec x) = \sum_{i=1}^{N-1} k\left(|\vec r_{i+1}-\vec r_i| - r_0\right)^2 + \sum_{|i-j|\geq 2} 4\left[\left(\frac{1}{r_{ij}}\right)^{12} - \left(\frac{1}{r_{ij}}\right)^{6}\right],
\end{equation}
with $k = 50$, $r_0 = 1$, and $r_{ij} = |\vec r_i - \vec r_j|$.
The first term connects consecutive beads with harmonic bonds, which are fixed: they are never formed or broken.
The second is a Lennard-Jones interaction between all other pairs of beads.
The equilibrium distribution is $\pi(\vec x) \propto e^{-U(\vec x)/T}$, and the size of the chain is measured by its radius of gyration,
\begin{equation}
  R_g^2 = \frac{1}{N}\sum_{i=1}^{N}|\vec r_i - \vec r_\text{cm}|^2, \qquad \vec r_\text{cm} = \frac{1}{N}\sum_{i=1}^N \vec r_i.
\end{equation}

Starter code is available at \url{https://github.com/jmduarte/phys142/blob/main/assignments/final_projects/polymer_mcmc.py}.
It implements $U$, $R_g^2$, and a single-bead Metropolis sampler~\cite{Metropolis:1953} with a symmetric Gaussian proposal.
It is correct but slow.
For Problems B--D you will need to speed it up, e.g., by computing the energy change of a single-bead move in $O(N)$ rather than $O(N^2)$ and compiling the inner loop with Numba.
Report the speedup you achieve.

\begin{problem}
  \emph{Validation.}
  For $N=2$, the chain is a single bond.
  Show that the bond length $d$ is distributed as $P(d) \propto d^2 e^{-k(d-r_0)^2/T}$, and explain the origin of the factor $d^2$.
  Show that $\langle d\rangle \approx r_0 + T/(k r_0)$ to leading order in $T/(k r_0^2)$.
  At $T=1$, compute $\langle d \rangle$ by numerical quadrature and compare it with your MCMC estimate.
  Your MCMC error bar must account for autocorrelation between samples, e.g., using blocking~\cite{Flyvbjerg:1989} or the integrated autocorrelation time.
\end{problem}

\begin{problem}
  \emph{Coil--globule transition.}
  For $N=16$ and $N=32$, scan the temperature over $0.4 \leq T \leq 5$.
  Plot $\langle R_g^2\rangle^{1/2}$, the energy per bead $\langle U\rangle/N$, and the heat capacity per bead $C_V/N = (\langle U^2\rangle - \langle U\rangle^2)/(N T^2)$ versus $T$, all with error bars (use a jackknife or bootstrap for $C_V$).
  Estimate the temperature at which the chain collapses.
  Does $C_V$ have a clear peak at the collapse? Discuss.
  Describe your strategy for the initial configuration, burn-in, step size (and resulting acceptance rate), number of sweeps, and thinning (if any).
  Plot the integrated autocorrelation time of $R_g^2$ versus $T$ and discuss what it implies for how long you must run.
\end{problem}

\begin{problem}
  \emph{The Hastings correction.}
  Replace the symmetric proposal with a biased one: displace the chosen bead by $\vec\delta \sim \mathcal N(\mu\hat x, s^2 \mathbb{1})$ with $\mu = s$.
  Since $q(\vec x'|\vec x)\neq q(\vec x|\vec x')$, the acceptance probability must include the Hastings factor~\cite{Hastings:1970},
  \begin{equation}
    A(\vec x\to\vec x') = \min\left(1, e^{-\Delta U/T}\,\frac{q(\vec x|\vec x')}{q(\vec x'|\vec x)}\right).
  \end{equation}
  Derive $q(\vec x|\vec x')/q(\vec x'|\vec x)$ for this proposal.
  For $N=16$ at $T=1$, compare $\langle R_g^2\rangle$ and the diagonal components $\langle S_{xx}\rangle$, $\langle S_{yy}\rangle$, $\langle S_{zz}\rangle$ of the gyration tensor, $S_{\alpha\beta} = \frac{1}{N}\sum_i (r_{i,\alpha} - r_{\text{cm},\alpha})(r_{i,\beta} - r_{\text{cm},\beta})$, from three samplers: symmetric, biased without the Hastings factor, and biased with it.
  Which agree with each other? What happens to the acceptance rate and the autocorrelation time?
  Finally, show that for $N=2$ the biased sampler gives the correct $\langle d\rangle$ even without the Hastings factor, and explain why.

  \begin{hint}
    Compare averages, e.g., $\langle S_{xx}\rangle$ with $\langle S_{yy}\rangle$, rather than the average of a ratio such as $\langle S_{xx}/S_{yy}\rangle$.
  \end{hint}
\end{problem}

\begin{problem}
  \emph{Scaling.}
  At a high temperature ($T=4$, swollen coil) and a low temperature ($T=0.4$, compact globule), measure $\langle R_g^2\rangle^{1/2}$ for $N = 8$, 16, and 32, and fit $R_g \propto N^\nu$.
  Compare your exponents with $\nu \approx 0.588$ for a self-avoiding walk, $\nu = 1/2$ for an ideal chain (the $\theta$ point), and $\nu = 1/3$ for a compact globule~\cite{deGennes:1979}.
  How does the autocorrelation time grow with $N$?
  Discuss finite-size effects and the limitations of single-bead moves.
  Including $N=64$ is optional and will likely require one of the extensions below.
\end{problem}

\noindent\textbf{Possible extensions:}
\begin{itemize}
  \item Pivot moves: rotate the part of the chain on one side of a randomly chosen bead by a random rotation~\cite{Madras:1988}.
        Show that this proposal is symmetric, and compare autocorrelation times with single-bead moves, especially for large $N$ at high $T$.
  \item Parallel tempering at low temperatures, where single-bead moves get stuck.
  \item Heteropolymers: make the Lennard-Jones well depth depend on the bead types (e.g., hydrophobic and polar beads), and study whether a specific compact structure emerges.
  \item Start from the $C_\alpha$ trace of a real protein from the Protein Data Bank, and discuss why a homopolymer model does not keep it folded.
\end{itemize}
