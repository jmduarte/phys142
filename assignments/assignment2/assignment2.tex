\end{itemize}

\section{Tunneling in the Double Potential Well [40 Points + 10 Bonus Points]}
\materials{Lectures 10--11}

\textbf{Potential.} For a particle at position $x$, set $m=\hbar=1$ and use
\begin{equation}
  V(x) = \alpha x^4 - 2x^2 + \frac{1}{\alpha},
\end{equation}
The minima are at $x = \pm x_\mathrm{min}$, where $x_\mathrm{min} = 1/\sqrt{\alpha}$ and $V(\pm x_\mathrm{min}) = 0$.
The barrier height is $V(0) = 1/\alpha$; the figure uses $\alpha = 0.4$.
\begin{center}
  \includegraphics[width=0.5\textwidth]{V_all.pdf}
\end{center}

\textbf{Your parameter.} Let $d$ be the last three digits (000--999) of your Triton Student Number (TSN: nine digits starting with 2). Use
\begin{equation}
  \alpha = 0.3 + \frac{d}{5000},
\end{equation}
so $0.3 \leq \alpha < 0.5$.
Examples: TSN 212345678 gives $\alpha = 0.3 + 678/5000 = 0.4356$; TSN 212345123 gives $\alpha = 0.3246$.
State $\alpha$ at the top of your report and use it unless other values are specified.

\textbf{Harmonic approximation.} Since $V''(\pm x_\mathrm{min}) = 8$ for every $\alpha$, the approximate wells near each minimum are
\begin{align}
  V_+(x) & = 4(x-x_\mathrm{min})^2, \\
  V_-(x) & = 4(x+x_\mathrm{min})^2,
\end{align}
with $\omega = 2\sqrt{2}$ and ground states
\begin{equation}
  \psi_\pm(x) = \left(\frac{\omega}{\pi}\right)^{1/4} \exp\left(- \frac{\omega}{2}(x\mp x_\mathrm{min})^2\right),
\end{equation}
Their symmetric and antisymmetric combinations,
\begin{equation}
  \psi_\mathrm{S}(x) = \frac{1}{\sqrt{2}}\left( \psi_+(x) + \psi_-(x) \right), \qquad
  \psi_\mathrm{A}(x) = \frac{1}{\sqrt{2}}\left( \psi_+(x) - \psi_-(x) \right)
\end{equation}
approximate the ground state $\phi_0$ and first excited state $\phi_1$, whose energy gap is $\Delta E = E_1-E_0$.

\Needspace{8\baselineskip}
\begin{problem}[10]
  \textbf{Path-integral tunneling}

  Start in the right well, $\psi(x,0)=\psi_+(x)$, and propagate one time step at a time with
  \begin{equation}
    \tilde{\mathcal K}(x_b, \epsilon; x_a, 0) = \exp \left[ i \left(\frac{1}{2}\frac{(x_b - x_a)^2}{\epsilon} - V\left(\frac{x_a+x_b}{2}\right) \epsilon\right) \right],
  \end{equation}
  where $x_a$ and $x_b$ are the initial and final positions.
  The autograder requires this midpoint potential.
  The propagator omits its normalization factor, so normalize after every step:
  \begin{equation}
    \psi(x, t) \leftarrow \frac{\psi(x, t)}{\sqrt{\int |\psi(x, t)|^2\, dx}}.
  \end{equation}

  \begin{enumerate}[(i)]
    \item \textbf{Run and plot.} Follow one full slow oscillation, from right to left and back.

\end{problem}

\begin{problem}[10]
  \textbf{Numerical eigenstates}

  Find the two lowest eigenvalues $E_0,E_1$ and eigenfunctions $\phi_0,\phi_1$ of $\hat H=-\frac12\frac{\partial^2}{\partial x^2}+V(x)$ (Lecture 11).
  Use the $(N_D+1)\times(N_D+1)$ centered three-point matrix
  \begin{equation}
    H_{ii}=\frac{1}{\Delta x^2}+V(x_i), \qquad
    H_{i,i+1}=H_{i+1,i}=-\frac{1}{2\Delta x^2},
  \end{equation}
  with all other entries zero.

  \textbf{Endpoints:} retain all $N_D+1$ unknowns, including both endpoints.
  Omit the outside neighbor in the first and last rows: this sets $\psi=0$ at $x_0-\Delta x$ and $x_{N_D}+\Delta x$, not at the endpoints themselves.
